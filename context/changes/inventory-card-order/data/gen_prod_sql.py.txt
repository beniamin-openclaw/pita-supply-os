"""Write prod-sql.sql (inventory-card-order) from positions.json + card_order.json.

Pipeline (re-run all three after any card change; never hand-edit
inventory_order on prod):
  1. extract_card_order.py.txt  -> card_order.json   (parse the 12 DRUK workbooks)
  2. gen_positions.py.txt       -> positions.json    (template + overrides, proven)
  3. gen_prod_sql.py.txt        -> ../prod-sql.sql   (this file)

Run from context/changes/inventory-card-order/data/:
  python3 gen_prod_sql.py.txt
"""
import json
import sys

STEP_FILE = "../prod-sql.sql"
cards = json.load(open("card_order.json"))["locations"]
pos = json.load(open("positions.json"))
template = pos["template"]            # [[product_id, position], ...] in card order
overrides = pos["overrides"]          # {location_id: {product_id: position}}
glob = dict(template)


def card_seq(loc):
    out, seen = [], set()
    for row in cards[loc]["rows"]:
        for pid in row["product_ids"]:
            if pid not in seen:
                seen.add(pid)
                out.append(pid)
    return out


seqs = {loc: card_seq(loc) for loc in sorted(cards)}

# Re-prove gen_positions' invariant: sorting each card by (effective, product_id)
# reproduces it. Fail loudly rather than emit SQL that breaks a card.
for loc, seq in seqs.items():
    ov = overrides.get(loc, {})
    eff = {p: ov.get(p, glob.get(p)) for p in seq}
    if any(v is None for v in eff.values()):
        sys.exit(f"{loc}: a card product has no position")
    if sorted(seq, key=lambda p: (eff[p], p)) != seq:
        sys.exit(f"{loc}: template + overrides do not reproduce the card")

name_by_pid = {}
for loc in cards:
    for row in cards[loc]["rows"]:
        for pid in row["product_ids"]:
            name_by_pid.setdefault(pid, row["name"])


def sql_str(s):
    return "'" + s.replace("'", "''") + "'"


def values(rows, indent="    "):
    return (",\n").join(indent + "(" + ", ".join(r) + ")" for r in rows)


template_values = values(
    [[sql_str(p), str(v)] for p, v in template]
)
override_rows = [
    [sql_str(loc), sql_str(p), str(v)]
    for loc in sorted(overrides)
    for p, v in sorted(overrides[loc].items(), key=lambda kv: kv[1])
]
override_values = values(override_rows)
card_values = values(
    [[sql_str(loc), str(i + 1), sql_str(p)] for loc, seq in seqs.items() for i, p in enumerate(seq)]
)
n_template = len(template)
n_overrides = len(override_rows)
override_counts = ", ".join(f"{loc} {len(v)}" for loc, v in sorted(overrides.items()) if v)

CARD_CTE = f"card(location_id, seq, product_id) AS (VALUES\n{card_values}\n)"

# Effective position of every setting, and its rank in the location's list.
EFF_CTE = """eff AS (
    SELECT s.location_id, s.product_id, p.product_category, p.active,
           COALESCE(s.inventory_order, p.inventory_order) AS e
    FROM location_product_settings s
    JOIN products p ON p.product_id = s.product_id
)"""

out = []
w = out.append
w(f"""-- inventory-card-order — prod data step: inventory-card positions
-- Supabase project lpzhphufjwrndfogkfub.
-- GENERATED by data/gen_prod_sql.py.txt from data/positions.json and
-- data/card_order.json. Do not edit by hand; re-run the pipeline instead.
--
-- Run ONLY with the operator's explicit approval, and only AFTER:
--   1. migration 0027 is applied on prod (list_migrations shows it), and
--   2. the PR is merged and the new Railway backend + Vercel bundle are live, and
--   3. the 12 inventory workbooks were re-downloaded and the pipeline re-run
--      (the cards were still being edited before 1.10); commit any change first.
-- Lessons: diff before (the diff IS the rollback), apply, audit after.
--
-- Positions are OWNED BY THE PIPELINE. Never hand-edit inventory_order on prod:
-- step 2 clears every value and re-derives it, so a hand-set value is lost on
-- the next run. Everything keys on (location_id, product_id) / product_id —
-- never on setting_id, never on a name.
--
-- What it writes:
--   products.inventory_order                   the common card template
--                                              ({n_template} products, step 10)
--   location_product_settings.inventory_order  per-location overrides that
--                                              reproduce each card ({n_overrides} rows)
--                                              + "extras": a location product its
--                                              own card does not list goes to the
--                                              end of its category section
--                                              (section's last card position + 1).
-- Effective position = override, else template; ties by product_id
-- (app/product_order.py). Expected overrides per location: {override_counts}.

-- ---------- 1. Diff before (save the output as prod-sql-diff-before.md — it is the rollback) ----------

-- 1a. Current values that step 2 will change (all NULL on the first run).
SELECT 'product' AS kind, NULL AS location_id, product_id, inventory_order
FROM products WHERE inventory_order IS NOT NULL
UNION ALL
SELECT 'setting', location_id, product_id, inventory_order
FROM location_product_settings WHERE inventory_order IS NOT NULL
ORDER BY 1, 2, 3;
-- expect on the first run: 0 rows

-- 1b. Card products with NO setting at their location (not on that grid;
--     informational — the card lists something the app does not stock there).
WITH {CARD_CTE}
SELECT c.location_id, c.seq, c.product_id
FROM card c
LEFT JOIN location_product_settings s
       ON s.location_id = c.location_id AND s.product_id = c.product_id
WHERE s.setting_id IS NULL
ORDER BY 1, 2;

-- 1c. Overrides that match no setting (must be 0 rows).
SELECT o.location_id, o.product_id
FROM (VALUES
{override_values}
) AS o(location_id, product_id, pos)
LEFT JOIN location_product_settings s
       ON s.location_id = o.location_id AND s.product_id = o.product_id
WHERE s.setting_id IS NULL;
-- expect: 0 rows

-- 1d. Extras per location: active location products their own card does not
--     list, which step 2 moves to the end of their category section.
WITH {CARD_CTE}
SELECT s.location_id, p.product_category, s.product_id, p.product_name_pl
FROM location_product_settings s
JOIN products p ON p.product_id = s.product_id AND p.active
WHERE s.location_id IN (SELECT DISTINCT location_id FROM card)
  AND NOT EXISTS (SELECT 1 FROM card c
                  WHERE c.location_id = s.location_id AND c.product_id = s.product_id)
ORDER BY 1, 2, 3;

-- ---------- 2. Apply (one transaction) ----------

BEGIN;

-- 2a. Clear everything first so a re-run is exact.
UPDATE products SET inventory_order = NULL WHERE inventory_order IS NOT NULL;
UPDATE location_product_settings SET inventory_order = NULL WHERE inventory_order IS NOT NULL;

-- 2b. Template.
UPDATE products AS p
SET inventory_order = t.pos
FROM (VALUES
{template_values}
) AS t(product_id, pos)
WHERE p.product_id = t.product_id;
-- expect: UPDATE {n_template} (fewer only if a product id is missing on prod)

-- 2c. Card overrides.
UPDATE location_product_settings AS s
SET inventory_order = o.pos
FROM (VALUES
{override_values}
) AS o(location_id, product_id, pos)
WHERE s.location_id = o.location_id AND s.product_id = o.product_id;
-- expect: UPDATE {n_overrides}

-- 2d. Extras: section end + 1, computed from the card's effective positions and
--     prod's product_category (ties among extras break by product_id).
--     "+ 1" relies on a gap of at least 2 before the next section's first
--     position (template and overrides step by 10). Audit 3d is the tripwire:
--     it fails when a future override lands at section end + 1.
--     Settings of inactive products are positioned too, on purpose: they never
--     render, and a re-activated product already sits at its section end.
WITH {CARD_CTE},
card_eff AS (
    SELECT c.location_id, p.product_category,
           COALESCE(s.inventory_order, p.inventory_order) AS e
    FROM card c
    JOIN products p ON p.product_id = c.product_id
    LEFT JOIN location_product_settings s
           ON s.location_id = c.location_id AND s.product_id = c.product_id
),
section_end AS (
    SELECT location_id, product_category, max(e) AS se
    FROM card_eff GROUP BY 1, 2
)
UPDATE location_product_settings AS s
SET inventory_order = se.se + 1
FROM products p, section_end se
WHERE p.product_id = s.product_id
  AND se.location_id = s.location_id
  AND se.product_category = p.product_category
  AND NOT EXISTS (SELECT 1 FROM card c
                  WHERE c.location_id = s.location_id AND c.product_id = s.product_id);
-- expect: UPDATE = the extras listed in 1d whose category has a section on the
--         card, plus inactive products' settings (82 on the 2026-09-28 prod copy)

COMMIT;

-- ---------- 3. Audit after ----------

-- 3a. Template: every template product positioned, all positions > 0.
SELECT count(*) FILTER (WHERE inventory_order IS NOT NULL) AS positioned,
       count(*) FILTER (WHERE inventory_order <= 0) AS non_positive
FROM products;
-- expect: {n_template} | 0

-- 3b. Card overrides applied per location (extras excluded).
WITH {CARD_CTE}
SELECT s.location_id, count(*) AS card_overrides
FROM location_product_settings s
JOIN card c ON c.location_id = s.location_id AND c.product_id = s.product_id
WHERE s.inventory_order IS NOT NULL
GROUP BY 1 ORDER BY 1;
-- expect: {override_counts} (locations with 0 are absent)

-- 3c. Card order holds: at every location, the card products it stocks, sorted
--     by (effective position, product_id), come out in card sequence.
WITH {CARD_CTE},
{EFF_CTE},
ranked AS (
    SELECT c.location_id, c.seq,
           lag(c.seq) OVER (PARTITION BY c.location_id
                            ORDER BY e.e NULLS LAST, e.product_id) AS prev_seq
    FROM card c
    JOIN eff e ON e.location_id = c.location_id AND e.product_id = c.product_id
)
SELECT location_id, count(*) AS inversions
FROM ranked WHERE prev_seq > seq
GROUP BY 1 ORDER BY 1;
-- expect: 0 rows

-- 3d. Contiguity: no active non-card product sorts between two card products
--     of the same category section.
WITH {CARD_CTE},
{EFF_CTE},
listed AS (
    SELECT e.*, row_number() OVER (PARTITION BY e.location_id
                                   ORDER BY e.e NULLS LAST, e.product_id) AS rn,
           EXISTS (SELECT 1 FROM card c
                   WHERE c.location_id = e.location_id AND c.product_id = e.product_id) AS on_card
    FROM eff e
    WHERE e.active AND e.location_id IN (SELECT DISTINCT location_id FROM card)
),
sections AS (
    SELECT location_id, product_category, min(rn) AS lo, max(rn) AS hi
    FROM listed WHERE on_card GROUP BY 1, 2
)
SELECT l.location_id, sec.product_category, l.product_id
FROM listed l
JOIN sections sec ON sec.location_id = l.location_id AND l.rn BETWEEN sec.lo AND sec.hi
WHERE NOT l.on_card
ORDER BY 1, 2, 3;
-- expect: 0 rows

-- 3e. Unpositioned active location products per location. Every active product
--     at a carded location is either on its card or an extra (2d), so this is
--     empty unless a product's category has no section on the card and no card
--     lists it anywhere; such rows sort last.
{'WITH ' + EFF_CTE}
SELECT location_id, count(*) AS unpositioned, string_agg(product_id, ',' ORDER BY product_id) AS ids
FROM eff WHERE active AND e IS NULL
GROUP BY 1 ORDER BY 1;
-- expect: 0 rows (local dry run on a prod copy, 2026-09-28)

-- 3f. Other suppliers' display_order untouched (this file never writes it).
SELECT supplier_id, count(*) FILTER (WHERE display_order IS NOT NULL) AS positioned
FROM supplier_products GROUP BY 1 HAVING count(*) FILTER (WHERE display_order IS NOT NULL) > 0
ORDER BY 1;
-- expect: same as before this file (SUP_BUKAT 14, plus Pago/Mory if
--         prod-sql-pago-mory.sql already ran)

-- ---------- R. Rollback ----------
-- First run: every value was NULL (step 1a returned 0 rows), so:
--   UPDATE products SET inventory_order = NULL WHERE inventory_order IS NOT NULL;
--   UPDATE location_product_settings SET inventory_order = NULL WHERE inventory_order IS NOT NULL;
-- Later runs: restore the rows saved from step 1a.
""")

open(STEP_FILE, "w").write("".join(out))
print(f"wrote {STEP_FILE}: template {n_template}, overrides {n_overrides}, "
      f"card rows {sum(len(s) for s in seqs.values())}")
